import tkinter as tk
from tkinter import filedialog, messagebox
import subprocess

def run_conversion():
    in_dir = filedialog.askdirectory(title="Select Word Input Folder")
    out_dir = filedialog.askdirectory(title="Select Output Folder")

    if not in_dir or not out_dir:
        return

    subprocess.run([
        "python",
        "word_to_edgar_enterprise.py",
        in_dir,
        out_dir
    ])

    messagebox.showinfo("Done", "EDGAR HTML generated successfully")

root = tk.Tk()
root.title("Word → EDGAR Converter")
root.geometry("400x150")

tk.Button(root, text="Convert Word to EDGAR", command=run_conversion,
          font=("Arial", 12), width=30).pack(pady=40)

root.mainloop()
